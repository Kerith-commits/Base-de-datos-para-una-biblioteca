CREATE TABLE Editoriales (
    Editorial_id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    Nombre VARCHAR(150) NOT NULL
);

-- 2. TABLA: Autores
CREATE TABLE Autores (
    Autor_id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    Nombre VARCHAR(150) NOT NULL
);

-- 3. TABLA: Libros (Tabla Central)
CREATE TABLE Libros (
    Libro_id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    Titulo VARCHAR(200) NOT NULL,
    Estado VARCHAR(50) DEFAULT 'Disponible',
    Autor_id BIGINT REFERENCES Autores(Autor_id) ON DELETE SET NULL,
    Editorial_id BIGINT REFERENCES Editoriales(Editorial_id) ON DELETE SET NULL
);

-- 4. TABLA: Categorias
CREATE TABLE Categorias (
    Categoria_id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    Nombre VARCHAR(100) NOT NULL,
    Libro_id BIGINT REFERENCES Libros(Libro_id) ON DELETE CASCADE
);

-- 5. TABLA: Prestamos
CREATE TABLE Prestamos (
    Prestamo_id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    Estado VARCHAR(50) DEFAULT 'Activo',
    Fecha_Prestamo TIMESTAMPTZ DEFAULT NOW(),
    Libro_id BIGINT REFERENCES Libros(Libro_id) ON DELETE CASCADE
);

-- 6. TABLA: Existencias
CREATE TABLE Existencias (
    Existencia_id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    Cantidad INT NOT NULL DEFAULT 0 CHECK (Cantidad >= 0),
    Libro_id BIGINT REFERENCES Libros(Libro_id) ON DELETE CASCADE
);
